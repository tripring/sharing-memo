```sql
WITH seq_info AS (
  SELECT
    n.nspname as schema_name,
    seq.relname as sequence_name,
    t.relname as table_name,
    a.attname as column_name
  FROM pg_class seq
    JOIN pg_namespace n ON n.oid = seq.relnamespace
    JOIN pg_depend d ON d.objid = seq.oid
    JOIN pg_class t ON t.oid = d.refobjid
    JOIN pg_attribute a ON a.attrelid = t.oid AND a.attnum = d.refobjsubid
  WHERE seq.relkind = 'S'
    AND d.refclassid = 'pg_class'::regclass
    AND d.classid = 'pg_class'::regclass
    AND d.deptype = 'a'
)
SELECT
  s.schema_name,
  s.sequence_name,
  s.table_name,
  s.column_name,
  max_col.max_val AS column_max_value,
  seq_val.last_value AS sequence_value
FROM
  seq_info s
  JOIN LATERAL (
    -- カラム最大値をSQL生成して取得
    EXECUTE format('SELECT MAX(%I) AS max_val FROM %I.%I', s.column_name, s.schema_name, s.table_name)
  ) AS max_col(max_val) ON TRUE
  JOIN LATERAL (
    -- シーケンス値
    SELECT last_value FROM pg_catalog.pg_sequences
    WHERE schemaname = s.schema_name AND sequencename = s.sequence_name
    LIMIT 1
  ) AS seq_val(last_value) ON TRUE
WHERE
  seq_val.last_value < max_col.max_val
;
```
